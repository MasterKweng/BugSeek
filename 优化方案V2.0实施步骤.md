  📋 对优化方案V2.0的理解

  🎯 核心战略思想

  "用算力换内存，用Postgres插件换组件，用图算法换AI消耗"

  这是一个针对 AMD Ryzen 7 7840HS + 16/32GB共享内存 环境的轻量化、高性价比重构方案。核心目标是：

   1. 解决内存瓶颈：不引入Milvus、Neo4j等重型中间件
   2. 提升准确率：通过图算法+上下文感知，从40-50%提升到85-90%
   3. 利用CPU优势：充分利用Zen4架构的高单核性能
   4. 控制风险：通过并发控制避免OOM和CPU卡死

  五阶段重构

  阶段1：全息数据提取
   - 问题：递归缺失（#3）、扁平键（#2）、逻辑重复（#4）
   - 方案：
     - 递归提取器：自动处理 array.items 和 object.properties
     - 逻辑键升级：{api_group}::{method}::{path}::{json_path}
     - 统一处理：Query参数解析集中化

  阶段2：基于图谱的候选生成
   - 问题：批次重心（#1）、字段驱动（#11）、无关系图（#12）、并发打爆（#13）、向量索引无缓存（#14）
   - 方案：
     - 并发控制：Semaphore(4) 限制并发
     - API级上下文：每次处理一个API的所有字段
     - 锚点表检测：通过度中心性+频次选出主表
     - 图簇过滤：2跳内的邻居表构成"合法业务域"

  阶段3：加权概率评分
   - 问题：score混用（#5）、强行加分（#6）、粗暴Stopword（#7）、高频字段（#10）
   - 方案：
     - 可解释公式：0.45*向量分 + 0.25*精确分*词权重 + 0.3*图谱分
     - TF-IDF思想：泛词权重低，专词权重高
     - 严格验证：必须在连通子图中

  阶段4：Schema-Aware AI优化
   - 问题：自动确认危险（#8）、锚定Top3（AI-2）、缺乏兄弟上下文（AI-3）
   - 方案：
     - 收紧网关：final_score > 0.9 且 s_graph == 1.0
     - 去除锚定：不发Top3候选，只发API结构和图簇范围
     - 兄弟节点上下文：让AI理解字段间的语义关系

  阶段5：语义验证
   - 问题：Jaccard相似度不准确（AI-4）
   - 方案：
     - 向量验证：使用Cosine Similarity
     - 阈值控制：similarity > 0.75 才通过

   def _extract_recursive(schema: dict, path: str = "") -> List[Tuple[str, dict]]:
       """
       递归提取所有字段路径

       Args:
           schema: JSON Schema
           path: 当前路径（前缀）

       Returns:
           [(field_path, field_schema), ...]
       """
       fields = []

       if not isinstance(schema, dict):
           return fields

       # 处理对象类型
       if schema.get('type') == 'object' and 'properties' in schema:
           for prop_name, prop_schema in schema['properties'].items():
               new_path = f"{path}.{prop_name}" if path else prop_name
               fields.extend(_extract_recursive(prop_schema, new_path))

       # 处理数组类型
       elif schema.get('type') == 'array' and 'items' in schema:
           new_path = f"{path}[]" if path else "[]"
           fields.extend(_extract_recursive(schema['items'], new_path))

       # 处理基础类型（叶子节点）
       else:
           fields.append((path, schema))

       return fields

    # app/field_mapping/graph_builder.py
  
    import networkx as nx
    from typing import Dict, Set, List, Optional
    from sqlalchemy import text
  
    class ForeignKeyGraph:
        """外键关系图"""
  
        def __init__(self, db_session):
            self.db = db_session
            self.graph = nx.Graph()
            self.table_columns = {}  # {table_name: [column_names]}
            self.foreign_keys = []   # [(from_table, from_col, to_table, to_col)]
  
        def build(self, db_schema: dict) -> None:
            """构建外键关系图（修正版，支持软外键推断）"""
            # 1. 从数据库读取外键信息
            self._load_foreign_keys_from_db()
  
            # 2. 如果没有外键，推断软外键（解决报错.md 问题1）
            if not self.foreign_keys:
                logger.warning("未找到物理外键，启用软外键推断")
                all_tables = list(db_schema.get('tables', {}).keys())
                self._infer_soft_foreign_keys(all_tables)
  
            # 3. 构建图
            for fk in self.foreign_keys:
                from_table, from_col, to_table, to_col = fk
                # 物理外键权重1.0，软外键权重0.5
                weight = 1.0 if self._is_physical_fk(from_table, to_table) else 0.5
                self.graph.add_edge(from_table, to_table, weight=weight)
  
            logger.info(f"外键图构建完成: {self.graph.number_of_nodes()}个表, {self.graph.number_of_edges()}个关系")
  
        def _load_foreign_keys_from_db(self) -> None:
            """从数据库加载外键信息"""
            query = text("""
                SELECT
                    tc.table_name AS from_table,
                    kcu.column_name AS from_column,
                    ccu.table_name AS to_table,
                    ccu.column_name AS to_column
                FROM information_schema.table_constraints AS tc
                JOIN information_schema.key_column_usage AS kcu
                    ON tc.constraint_name = kcu.constraint_name
                JOIN information_schema.constraint_column_usage AS ccu
                    ON ccu.constraint_name = tc.constraint_name
                WHERE tc.constraint_type = 'FOREIGN KEY'
                    AND tc.table_schema = 'public'
            """)
  
            result = self.db.execute(query)
            self.foreign_keys = [(row[0], row[1], row[2], row[3]) for row in result]
  
        def _infer_soft_foreign_keys(self, all_tables: List[str]) -> None:
            """
            推断软外键（解决报错.md 问题1）
            
            如果查不到物理外键，自动扫描表结构
            规则：如果表A有字段user_id且存在表auth_user，则添加软边（权重0.5）
            """
            # 获取所有表的列信息
            table_columns = self._get_all_table_columns()
  
            for table, columns in table_columns.items():
                for col in columns:
                    # 检查是否是 _id 结尾的字段
                    if col.endswith('_id'):
                        # 猜测目标表名
                        target_table = self._guess_target_table(col[:-3])
  
                        if target_table and target_table in all_tables:
                            # 添加软边（权重0.5，低于物理外键的1.0）
                            self.graph.add_edge(table, target_table, weight=0.5)
                            self.foreign_keys.append((table, col, target_table, ''))
                            logger.debug(f"推断软外键: {table}.{col} -> {target_table}")
  
        def _guess_target_table(self, field_name: str) -> Optional[str]:
            """
            根据字段名猜测目标表名
            
            例如: user -> auth_user, part -> part_part
            """
            # 常见表名映射
            mappings = {
                'user': 'auth_user',
                'part': 'part_part',
                'order': 'sales_order',
                'customer': 'company_company',
                'supplier': 'company_company',
                'stock': 'stock_stockitem',
                'purchase': 'purchase_order',
                'invoice': 'sales_invoice',
            }
  
            # 直接映射
            if field_name in mappings:
                return mappings[field_name]
  
            # 尝试组合：{field_name}_{field_name}
            candidate = f"{field_name}_{field_name}"
            if candidate in self.graph.nodes():
                return candidate
  
            return None
  
        def _get_all_table_columns(self) -> Dict[str, List[str]]:
            """获取所有表的列信息"""
            table_columns = {}
            
            query = text("""
                SELECT table_name, column_name
                FROM information_schema.columns
                WHERE table_schema = 'public'
                ORDER BY table_name, ordinal_position
            """)
            
            result = self.db.execute(query)
            for row in result:
                table_name = row[0]
                column_name = row[1]
                
                if table_name not in table_columns:
                    table_columns[table_name] = []
                table_columns[table_name].append(column_name)
            
            return table_columns
  
        def _is_physical_fk(self, from_table: str, to_table: str) -> bool:
            """检查是否是物理外键"""
            for fk in self.foreign_keys:
                if fk[0] == from_table and fk[2] == to_table:
                    return True
            return False
  
        def get_neighbors(self, table: str, hops: int = 2) -> Set[str]:
            """获取指定跳数内的邻居表"""
            if table not in self.graph:
                return set()
  
            neighbors = set()
            for node in self.graph.nodes():
                try:
                    distance = nx.shortest_path_length(self.graph, source=table, target=node)
                    if distance <= hops:
                        neighbors.add(node)
                except nx.NetworkXNoPath:
                    continue
  
            return neighbors
  
        def get_degree_centrality(self) -> Dict[str, float]:
            """获取所有表的度中心性"""
            return nx.degree_centrality(self.graph)
  
    步骤2.2：实现锚点表检测（含置信度阈值）
    # app/field_mapping/processor.py
  
    def _detect_anchor_table(
        self,
        all_candidates: List[List[Dict]],
        graph: ForeignKeyGraph
    ) -> Optional[str]:
        """
        检测锚点表（Anchor Table，修正版，解决报错.md 问题2）
        
        算法：
        1. 统计所有候选表的出现频次
        2. 计算候选表的度中心性
        3. 选择 频次*中心性 最高的表作为锚点
        4. 增加置信度阈值，避免零结果风险
        """
        table_freq = {}
        table_scores = {}
  
        # 1. 统计频次
        for candidates in all_candidates:
            for cand in candidates[:5]:  # 只看Top5
                table = cand.get('db_table', '')
                if table:
                    table_freq[table] = table_freq.get(table, 0) + 1
  
        if not table_freq:
            logger.warning("没有候选表，跳过锚点检测")
            return None
  
        # 2. 计算度中心性
        centrality = graph.get_degree_centrality()
  
        # 3. 综合评分
        for table, freq in table_freq.items():
            degree = centrality.get(table, 0)
            table_scores[table] = freq * (1 + degree * 10)  # 度中心性权重更高
  
        # 4. 选择最高分
        max_score = max(table_scores.values())
        anchor_table = max(table_scores, key=table_scores.get)
  
        # 5. 置信度检查（新增，解决报错.md 问题2）
        confidence_threshold = 2.0  # 阈值可调整
        if max_score < confidence_threshold:
            logger.warning(f"锚点表置信度不足 ({max_score:.2f} < {confidence_threshold})，回退到纯向量匹配")
            return None
  
        # 6. 检查分散度（新增，解决报错.md 问题2）
        scores_list = list(table_scores.values())
        if len(scores_list) > 1:
            avg_score = sum(scores_list) / len(scores_list)
            if max_score / avg_score < 1.5:  # 最高分与平均分的比例
                logger.warning(f"候选表过于分散 (max/avg={max_score/avg_score:.2f})，回退到纯向量匹配")
                return None
  
        logger.info(f"检测到锚点表: {anchor_table} (置信度={max_score:.2f})")
        return anchor_table

       for result in results:
           field_registry.update(result)

       return field_registry

    import yaml
    from collections import Counter
  
    class TokenWeightCalculator:
        """词权重计算器（配置化版本，解决报错.md 问题5）"""
        
        def __init__(self, config_path: str = "config/token_weights.yaml"):
            self.common_tokens = set()

        def _load_config(self, config_path: str) -> None:
            """从配置文件加载词库"""
            try:
                with open(config_path, 'r', encoding='utf-8') as f:
                    config = yaml.safe_load(f)
                    
                self.common_tokens = set(config.get('common_tokens', []))
                self.specific_tokens = set(config.get('specific_tokens', []))
                
                logger.info(f"词权重配置加载完成: {len(self.common_tokens)}个通用词, {len(self.specific_tokens)}个专有词")
            except Exception as e:
                logger.warning(f"加载词权重配置失败，使用默认值: {e}")
                # 使用默认值（降级策略）
                self.common_tokens = {
                    'id', 'name', 'code', 'type', 'status', 'time', 'date',
                    'user', 'created', 'updated', 'deleted', 'description'
                }
                self.specific_tokens = {
                    'serial', 'part', 'order', 'invoice', 'purchase', 'stock',
                    'customer', 'supplier', 'category', 'brand', 'model'
                }
        
        def get_token_weight(self, token: str) -> float:
            """
            获取词权重
  
            - 通用词: 0.2
            - 专有词: 1.0
            - 其他: 0.6
            """
            token_lower = token.lower()
            
            if token_lower in self.common_tokens:
                return 0.2
            elif token_lower in self.specific_tokens:
                return 1.0
            else:
                return 0.6

       解决 #5 score混用, #6 强行加分, #7 粗暴Stopword, #10 高频字段
       """
       # 1. 向量基础分 (0.0 - 1.0)
       s_vector = cand.get('score', 0.0)

       # 2. 精确匹配分 (0.0 或 1.0)
       is_exact_match = cand.get('db_column', '').lower() == field_info.field_name.lower()
       s_exact = 1.0 if is_exact_match else 0.0

       # 3. 词权重（TF-IDF思想）
       token_weight = TokenWeightCalculator.get_token_weight(field_info.field_name)

       # 4. 图谱连通分 (0.0 或 1.0)
       candidate_table = cand.get('db_table', '')
       is_in_graph = candidate_table in graph_context.get('valid_tables', set())
       s_graph = 1.0 if is_in_graph else 0.0

       # 5. 加权求和
       final_score = (
           s_vector * 0.45 +
           s_exact * 0.25 * token_weight +
           s_graph * 0.30
       )

       # 6. 封顶
       return min(1.0, final_score)

   def _build_ai_prompt_v2(
       self,

           },

           # 注意：不发送Top3候选，让AI独立思考
       }

   def _get_sibling_fields(
       self,
       field_info: FieldInfo,
       api_context: Optional[APIContext] = None  # 新增参数，解决报错.md 问题3
   ) -> List[str]:
       """
       获取兄弟字段（修正版，解决报错.md 问题3）
       
       直接从API上下文获取，避免遍历注册表
       """
       if not api_context:
           logger.warning("缺少API上下文，无法获取兄弟字段")
           return []

       # 获取同一级别的字段
       base_path = field_info.field_path.rsplit('.', 1)[0]

       siblings = []
       for field in api_context.all_fields:
           if field.startswith(base_path) and field != field_info.field_path:
               siblings.append(field)

       return siblings
