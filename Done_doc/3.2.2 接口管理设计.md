# 3.2.2 接口定义管理 - 具体设计方案

## 一、功能概述

**功能定位**：管理解析后的接口定义，为后续测试脚本生成、场景组装、Mock 服务提供基础数据

**核心价值**：
- 统一接口定义管理，避免重复维护
- 支持多维度搜索和过滤，快速定位接口
- 支持接口分组和标签，便于组织和分类
- 支持接口编辑和补充，完善测试数据

## 五、核心功能实现逻辑

### 5.1 接口列表查询（支持多维度过滤）

```python
@router.get("/endpoints", response_model=ApiResponse)
async def get_endpoints(
    skip: int = 0,
    limit: int = 100,
    method: Optional[str] = None,
    tag: Optional[str] = None,
    group_id: Optional[int] = None,
    keyword: Optional[str] = None,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """获取接口列表，支持多维度过滤"""
    query = db.query(ApiEndpoint)
    
    # 方法过滤
    if method:
        query = query.filter(ApiEndpoint.method == method.upper())
    
    # 标签过滤（JSON 数组包含）
    if tag:
        query = query.filter(ApiEndpoint.tags.contains([tag]))
    
    # 分组过滤
    if group_id:
        query = query.filter(ApiEndpoint.group_id == group_id)
    
    # 关键词搜索（路径、摘要、描述）
    if keyword:
        query = query.filter(
            or_(
                ApiEndpoint.path.ilike(f"%{keyword}%"),
                ApiEndpoint.summary.ilike(f"%{keyword}%"),
                ApiEndpoint.description.ilike(f"%{keyword}%")
            )
        )
    
    total = query.count()
    endpoints = query.offset(skip).limit(limit).all()
    
    return ApiResponse(
        message="success",
        data={
            "endpoints": [EndpointResponse.from_orm(ep).model_dump() for ep in endpoints],
            "total": total
        }
    )
```

### 5.2 高级搜索接口

```python
@router.post("/endpoints/search", response_model=ApiResponse)
async def search_endpoints(
    request: EndpointSearchRequest,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """高级搜索接口"""
    query = db.query(ApiEndpoint)
    
    # 关键词搜索
    if request.keyword:
        query = query.filter(
            or_(
                ApiEndpoint.path.ilike(f"%{request.keyword}%"),
                ApiEndpoint.summary.ilike(f"%{request.keyword}%"),
                ApiEndpoint.description.ilike(f"%{request.keyword}%")
            )
        )
    
    # 方法过滤（多选）
    if request.methods:
        query = query.filter(ApiEndpoint.method.in_([m.upper() for m in request.methods]))
    
    # 标签过滤（多选，任意匹配）
    if request.tags:
        query = query.filter(
            or_(*[ApiEndpoint.tags.contains([tag]) for tag in request.tags])
        )
    
    # 分组过滤
    if request.group_id:
        query = query.filter(ApiEndpoint.group_id == request.group_id)
    
    # 路径模式匹配（正则表达式）
    if request.path_pattern:
        query = query.filter(ApiEndpoint.path.op('~')(request.path_pattern))
    
    endpoints = query.all()
    
    return ApiResponse(
        message="success",
        data={
            "endpoints": [EndpointResponse.from_orm(ep).model_dump() for ep in endpoints],
            "total": len(endpoints)
        }
    )
```

### 5.3 标签统计接口

```python
@router.get("/endpoints/tags", response_model=ApiResponse)
async def get_endpoint_tags(
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    """获取所有标签及使用次数"""
    # 查询所有接口的标签
    endpoints = db.query(ApiEndpoint).all()
    
    # 统计标签使用次数
    tag_stats = {}
    for ep in endpoints:
        if ep.tags:
            for tag in ep.tags:
                tag_stats[tag] = tag_stats.get(tag, 0) + 1
    
    # 按使用次数排序
    sorted_tags = sorted(tag_stats.items(), key=lambda x: x[1], reverse=True)
    
    return ApiResponse(
        message="success",
        data=[
            {"tag": tag, "count": count}
            for tag, count in sorted_tags
        ]
    )
```
