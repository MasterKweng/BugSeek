  5.3 核心解析流程

   # 文档解析流程
   def parse_document(document: ApiDocument) -> List[ApiEndpoint]:
       """
       1. 根据 source_type 选择解析器
       2. 解析文档内容
       3. 提取接口定义
       4. 保存到 api_endpoints 表
       5. 返回提取的接口数
       """
       parser = ParserFactory.create(document.source_type)
       endpoints = parser.parse(document.content)
       save_endpoints(endpoints, document.id)
       return endpoints
