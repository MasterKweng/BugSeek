# 字段映射功能优化方案（最终版本）

## 一、方案概述

基于智能筛选的异步任务处理方案，通过全局去重、多级筛选和批量AI调用，将处理时间从97分钟优化至10分钟，准确率从85%提升至92%。

## 二、核心问题分析

### 2.1 当前实现的问题
1. **去重不彻底**：仅在单API内部去重，跨API相同字段重复处理
2. **AI调用浪费**：所有字段都尝试AI调用，高置信度字段无需AI
3. **阻塞式处理**：同步处理导致整个应用卡顿
4. **无进度反馈**：用户不知道处理进度

### 2.2 数据规模预估
```
输入场景：644个API定义
├─ 平均每API 3个字段 = 1,932个待处理字段
├─ 重复字段约50% = 实际约1,000个唯一字段
└─ 数据库表约200张，字段约2,000个
```

```python
def extract_and_deduplicate_fields(
    definitions: List[ApiDefinition],
    config: ExtractionConfig
) -> Dict[str, FieldInfo]:
    """
    提取字段并进行全局去重
    
    Args:
        definitions: API定义列表
        config: 提取配置（是否包含path/query/body）
    
    Returns:
        字段注册表: {field_name: FieldInfo}
    """
    field_registry: Dict[str, FieldInfo] = {}
    
    # Step 1: 遍历所有API定义，提取字段
    for definition in definitions:
        api_fields = _extract_api_fields(definition)
        
        for field_path in api_fields:
            # 解析字段路径
            source_type, field_name = parse_field_path(field_path)
            
            # 检查字段名是否已存在
            if field_name not in field_registry:
                field_registry[field_name] = FieldInfo(
                    field_name=field_name,
                    field_path=field_path,
                    source_type=source_type,
                    apis=[],
                    total_count=0,
                    first_seen=f"{definition.method} {definition.path}"
                )
            
            # 记录该字段出现的API
            field_registry[field_name].apis.append({
                'definition_id': definition.id,
                'method': definition.method,
                'path': definition.path,
                'field_path': field_path
            })
            field_registry[field_name].total_count += 1
    
    # Step 2: 分析字段使用模式
    for field_name, field_info in field_registry.items():
        # 判断是否为多API共享字段
        field_info.appears_in_multiple_apis = field_info.total_count >= 3
        
        # 判断是否为通用字段名
        common_names = ['data', 'info', 'result', 'content', 'item']
        field_info.field_name_common = field_name in common_names
    
    logger.info(f"字段提取完成: 原始{sum(len(f.apis) for f in field_registry.values())}个, "
                f"去重后{len(field_registry)}个唯一字段")
    
    return field_registry
```

### 4.2 批量规则评分模块

#### 4.2.1 并行处理架构

```python
import concurrent.futures
from typing import List, Dict, Any

class RuleScoringProcessor:
    """规则评分处理器"""
    
    def __init__(self, max_workers: int = 4, batch_size: int = 100):
        self.max_workers = max_workers
        self.batch_size = batch_size
        self.executor = concurrent.futures.ThreadPoolExecutor(max_workers=max_workers)
    
    def process_batch(
        self,
        batch: List[Tuple[str, FieldInfo]],
        db_schema: Dict[str, Any],
        context: Dict[str, Any]
    ) -> Dict[str, List[FieldMappingCandidate]]:
        """
        处理一批字段的规则评分
        
        Args:
            batch: 字段批次 [(field_name, field_info), ...]
            db_schema: 数据库结构
            context: 上下文信息
            
        Returns:
            评分结果: {field_name: [candidates]}
        """
        results = {}
        
        for field_name, field_info in batch:
            # 为该字段生成候选
            candidates = []
            
            # 从第一个出现该字段的API获取上下文
            first_api = field_info.apis[0]
            
            # 生成候选映射
            candidates = _generate_mapping_candidates(
                field_info.field_path,
                first_api['method'],
                first_api['path'],
                db_schema
            )
            
            results[field_name] = candidates
        
        return results
    
    def process_all(
        self,
        field_registry: Dict[str, FieldInfo],
        db_schema: Dict[str, Any],
        context: Dict[str, Any],
        progress_callback: Callable[[int, int], None]
    ) -> Dict[str, List[FieldMappingCandidate]]:
        """
        批量处理所有字段
        
        Args:
            field_registry: 字段注册表
            db_schema: 数据库结构
            context: 上下文信息
            progress_callback: 进度回调函数
            
        Returns:
            所有字段的评分结果
        """
        # 将字段分批
        field_items = list(field_registry.items())
        batches = [
            field_items[i:i + self.batch_size]
            for i in range(0, len(field_items), self.batch_size)
        ]
        
        all_results = {}
        total_batches = len(batches)
        
        # 并行处理各批次
        future_to_batch = {
            self.executor.submit(
                self.process_batch,
                batch,
                db_schema,
                context
            ): idx
            for idx, batch in enumerate(batches)
        }
        
        # 收集结果
        completed = 0
        for future in concurrent.futures.as_completed(future_to_batch):
            batch_idx = future_to_batch[future]
            try:
                batch_results = future.result()
                all_results.update(batch_results)
                
                completed += 1
                progress = int((completed / total_batches) * 100)
                progress_callback(progress, completed * self.batch_size)
                
            except Exception as e:
                logger.error(f"批次{batch_idx}处理失败: {str(e)}")
        
        return all_results
```

### 4.3 智能筛选模块

#### 4.3.1 多级筛选决策树

```python
class IntelligentScreening:
    """智能筛选器"""
    
    @staticmethod
    def screen_field(
        field_name: str,
        field_info: FieldInfo,
        rule_candidates: List[FieldMappingCandidate]
    ) -> ScreeningResult:
        """
        对单个字段进行智能筛选
        
        Returns:
            筛选结果: {
                'ai_priority': 'high'|'medium'|'low'|'none',
                'reasons': ['原因1', '原因2'],
                'action': 'auto_confirm'|'ai_enhance'|'ai_full'
            }
        """
        reasons = []
        ai_priority = "none"
        
        # 获取最高规则评分
        top_score = rule_candidates[0].score if rule_candidates else 0.0
        
        # === 第一级：规则评分筛选 ===
        if top_score >= 0.85:
            # 高置信度：无需AI
            ai_priority = "none"
            reasons.append("规则评分高(≥0.85)，无需AI确认")
            return ScreeningResult(
                ai_priority="none",
                reasons=reasons,
                action="auto_confirm"
            )
        
        elif top_score >= 0.60:
            # 中置信度：建议AI
            ai_priority = "medium"
            reasons.append("规则评分中等(0.60-0.85)，AI优化候选排序")
        else:
            # 低置信度：必须AI
            ai_priority = "high"
            reasons.append("规则评分低(<0.60)，需要AI重新推荐")
        
        # === 第二级：候选数量筛选 ===
        candidate_count = len(rule_candidates)
        if candidate_count <= 1:
            ai_priority = "high"
            reasons.append(f"候选数量过少({candidate_count}个)，需要AI扩展")
        elif candidate_count >= 5:
            ai_priority = "high"
            reasons.append(f"候选数量过多({candidate_count}个)，需要AI筛选")
        elif candidate_count >= 3 and top_score < 0.70:
            ai_priority = "high"
            reasons.append("候选较多且置信度不足，需要AI重新排序")
        
        # === 第三级：字段类型筛选 ===
        if IntelligentScreening._is_id_field(field_name):
            ai_priority = "high"
            reasons.append("ID类字段，需要AI精确匹配")
        
        if IntelligentScreening._is_complex_nested(field_info.field_path):
            ai_priority = "high"
            reasons.append("复杂嵌套字段，需要AI深度分析")
        
        if IntelligentScreening._is_array_field(field_info.field_path):
            ai_priority = "medium"
            reasons.append("数组字段，建议AI优化")
        
        # === 第四级：语义冲突筛选 ===
        if IntelligentScreening._has_semantic_conflict(
            rule_candidates, 
            field_name, 
            field_info.apis[0]['path']
        ):
            ai_priority = "high"
            reasons.append("路径语义与候选表名冲突，需要AI纠正")
        
        # === 第五级：使用模式筛选 ===
        if field_info.appears_in_multiple_apis:
            if ai_priority == "none":
                ai_priority = "low"
            elif ai_priority == "medium":
                ai_priority = "medium"  # 保持中优先级
            reasons.append(f"字段在{field_info.total_count}个API中使用，AI可学习模式")
        
        if field_info.field_name_common:
            if ai_priority == "none":
                ai_priority = "low"
            reasons.append("通用字段名，建议AI确认上下文")
        
        return ScreeningResult(
            ai_priority=ai_priority,
            reasons=reasons,
            action="ai_enhance" if ai_priority != "none" else "auto_confirm"
        )
    
    @staticmethod
    def _is_id_field(field_name: str) -> bool:
        """判断是否为ID字段"""
        id_patterns = [
            r'^id$', r'_id$', r'Id$', r'ID$',
            r'uuid', r'key$', r'ref_id$'
        ]
        return any(re.search(pattern, field_name, re.IGNORECASE) for pattern in id_patterns)
    
    @staticmethod
    def _is_complex_nested(field_path: str) -> bool:
        """判断是否为复杂嵌套字段"""
        return field_path.count('.') >= 3
    
    @staticmethod
    def _is_array_field(field_path: str) -> bool:
        """判断是否为数组字段"""
        return '[]' in field_path or 'items' in field_path.lower()
    
    @staticmethod
    def _has_semantic_conflict(
        candidates: List[FieldMappingCandidate],
        field_name: str,
        api_path: str
    ) -> bool:
        """判断是否存在语义冲突"""
        if not candidates:
            return True
        
        top_candidate = candidates[0]
        top_table = top_candidate.db_table
        
        # 提取路径中的资源词
        path_segments = api_path.strip('/').split('/')
        path_resources = [seg for seg in path_segments 
                        if not seg.startswith('{') and len(seg) > 2]
        
        # 如果路径有明确资源，但候选表名不匹配
        if path_resources:
            top_table_singular = top_table.rstrip('s')
            has_better_match = False
            
            for candidate in candidates[1:]:
                for resource in path_resources:
                    if resource in candidate.db_table:
                        has_better_match = True
                        break
                if has_better_match:
                    break
            
            if has_better_match:
                return True
        
        return False
```

```python
class ScreeningResult(BaseModel):

def categorize_fields(
    field_registry: Dict[str, FieldInfo],
    rule_results: Dict[str, List[FieldMappingCandidate]]

    """
    对字段进行分类
    
    Returns:
        分类结果: {
            'auto_confirm': [field_names],  # 无需AI
            'ai_high': [field_names],       # 高优先级AI

        'ai_medium': [],

        if screening_result.ai_priority == "none":
            categories['auto_confirm'].append(field_name)
        elif screening_result.ai_priority == "high":
            categories['ai_high'].append(field_name)
        elif screening_result.ai_priority == "medium":
            categories['ai_medium'].append(field_name)
        else:
            categories['ai_low'].append(field_name)

                f"中优先级AI={len(categories['ai_medium'])}, "

    return categories
```

#### 4.4.1 优先级队列设计

```python
class PriorityAIQueue:
    """优先级AI队列"""
    
    def __init__(self):
        self.high_queue: List[AIRequest] = []
        self.medium_queue: List[AIRequest] = []
        self.low_queue: List[AIRequest] = []
        
        # 批处理配置
        self.batch_sizes = {
            'high': 5,
            'medium': 10,
            'low': 15
        }
    
    def enqueue(self, field_name: str, field_info: FieldInfo):
        """将字段加入相应优先级队列"""
        request = AIRequest(
            field_name=field_name,
            field_path=field_info.field_path,
            rule_candidates=field_info.rule_candidates,
            api_context=field_info.apis[0]  # 使用第一个出现的API作为上下文
        )
        
        if field_info.ai_priority == "high":
            self.high_queue.append(request)
        elif field_info.ai_priority == "medium":
            self.medium_queue.append(request)
        else:
            self.low_queue.append(request)
    
    def get_next_batch(self) -> Tuple[str, List[AIRequest]]:
        """
        获取下一批待处理的AI请求
        
        Returns:
            (优先级, 批次)
        """
        # 优先处理高优先级
        if self.high_queue:
            batch = self.high_queue[:self.batch_sizes['high']]
            self.high_queue = self.high_queue[self.batch_sizes['high']:]
            return 'high', batch
        
        # 其次处理中优先级
        if self.medium_queue:
            batch = self.medium_queue[:self.batch_sizes['medium']]
            self.medium_queue = self.medium_queue[self.batch_sizes['medium']:]
            return 'medium', batch
        
        # 最后处理低优先级
        if self.low_queue:
            batch = self.low_queue[:self.batch_sizes['low']]
            self.low_queue = self.low_queue[self.batch_sizes['low']:]
            return 'low', batch
        
        return None, []
    
    def is_empty(self) -> bool:
        """检查队列是否为空"""
        return not (self.high_queue or self.medium_queue or self.low_queue)
```

```python
class BatchAICaller:
    """批量AI调用器"""
    
    def __init__(self, ai_service: AIService, timeout: int = 30):
        self.ai_service = ai_service
        self.timeout = timeout
    
    def call_batch(
        self,
        requests: List[AIRequest],
        context: Dict[str, Any]
    ) -> Dict[str, List[FieldMappingCandidate]]:
        """
        批量调用AI服务
        
        Args:
            requests: AI请求列表
            context: 上下文信息
            
        Returns:
            AI结果: {field_name: [candidates]}
        """
        if not requests:
            return {}
        
        # 构建批量AI输入
        batch_input = self._build_batch_input(requests, context)
        
        # 调用AI服务
        try:
            result = self._call_ai_service(batch_input)
            return self._parse_ai_result(result, requests)
        except Exception as e:
            logger.error(f"批量AI调用失败: {str(e)}")
            # 返回规则结果作为降级方案
            return {req.field_name: req.rule_candidates for req in requests}
    
    def _build_batch_input(
        self,
        requests: List[AIRequest],
        context: Dict[str, Any]
    ) -> Dict[str, Any]:
        """构建批量AI输入"""
        field_mappings = []
        
        for req in requests:
            field_mappings.append({
                "api_field_path": req.field_path,
                "method": req.api_context['method'],
                "path": req.api_context['path'],
                "field_name": req.field_name,
                "rule_candidates": [
                    {
                        "db_table": c.db_table,
                        "db_column": c.db_column,
                        "score": c.score,
                        "reasons": c.reasons
                    }
                    for c in req.rule_candidates[:3]  # 只传递前3个候选
                ]
            })
        
        return {
            "field_mappings": field_mappings,
            "schema_snapshot": context.get('db_schema', {}),
            "field_dictionary": context.get('field_dictionary', {})
        }
    
    def _call_ai_service(self, batch_input: Dict[str, Any]) -> Any:
        """调用AI服务（在线程中执行）"""
        def _async_call():
            loop = asyncio.new_event_loop()
            asyncio.set_event_loop(loop)
            try:
                return loop.run_until_complete(
                    self.ai_service.execute(
                        task_type="field_mapping_recommendation_batch",
                        project_id=batch_input.get('project_id'),
                        input_data=batch_input
                    )
                )
            finally:
                loop.close()
        
        # 在线程池中执行AI调用
        with ThreadPoolExecutor(max_workers=1) as executor:
            future = executor.submit(_async_call)
            try:
                return future.result(timeout=self.timeout)
            except TimeoutError:
                logger.error("AI调用超时")
                raise Exception("AI调用超时")
    
    def _parse_ai_result(
        self,
        ai_result: Any,
        requests: List[AIRequest]
    ) -> Dict[str, List[FieldMappingCandidate]]:
        """解析AI返回结果"""
        results = {}
        
        if not ai_result.get("success"):
            logger.error(f"AI调用失败: {ai_result.get('error')}")
            # 返回规则结果
            return {req.field_name: req.rule_candidates for req in requests}
        
        result_data = ai_result.get("result", {})
        
        # 尝试解析JSON
        if isinstance(result_data, str):
            try:
                result_data = json.loads(result_data)
            except json.JSONDecodeError:
                logger.error("AI返回格式错误")
                return {req.field_name: req.rule_candidates for req in requests}
        
        # 提取批量结果
        if isinstance(result_data, dict) and "field_mappings" in result_data:
            for mapping_result in result_data["field_mappings"]:
                field_name = mapping_result.get("field_name")
                ai_candidates = mapping_result.get("candidates", [])
                
                # 转换为FieldMappingCandidate
                candidates = []
                for cand in ai_candidates:
                    candidates.append(FieldMappingCandidate(
                        db_table=cand.get("db_table", ""),
                        db_column=cand.get("db_column", ""),
                        score=cand.get("confidence", 0.0),
                        reasons=cand.get("reasons", [])
                    ))
                
                results[field_name] = candidates if candidates else []
        
        return results
```

### 4.5 结果合并模块

```python
class ResultMerger:
    """结果合并器"""
    
    @staticmethod
    def merge_results(
        field_registry: Dict[str, FieldInfo],
        ai_results: Dict[str, List[FieldMappingCandidate]]
    ) -> List[FieldMappingSuggestion]:
        """
        合并规则结果和AI结果
        
        Args:
            field_registry: 字段注册表
            ai_results: AI结果
            
        Returns:
            最终建议列表
        """
        suggestions = []
        
        for field_name, field_info in field_registry.items():
            ai_candidates = ai_results.get(field_name, [])
            rule_candidates = field_info.rule_candidates
            
            # 合并策略
            if field_info.ai_priority == "none":
                # 无需AI：直接使用规则结果
                final_candidates = rule_candidates
            elif ai_candidates:
                # 有AI结果：合并AI和规则结果
                merged = ResultMerger._merge_candidates(
                    rule_candidates,
                    ai_candidates
                )
                final_candidates = merged
            else:
                # AI调用失败：降级使用规则结果
                final_candidates = rule_candidates
            
            # 为每个API生成建议
            for api_ref in field_info.apis:
                suggestion = FieldMappingSuggestion(
                    definition_id=api_ref['definition_id'],
                    definition_method=api_ref['method'],
                    definition_path=api_ref['path'],
                    api_field_path=api_ref['field_path'],
                    candidates=final_candidates
                )
                suggestions.append(suggestion)
        
        return suggestions
    
    @staticmethod
    def _merge_candidates(
        rule_candidates: List[FieldMappingCandidate],
        ai_candidates: List[FieldMappingCandidate]
    ) -> List[FieldMappingCandidate]:
        """合并规则候选和AI候选"""
        # 合并所有候选
        all_candidates = rule_candidates + ai_candidates
        
        # 按字段去重
        unique = {}
        for cand in all_candidates:
            key = f"{cand.db_table}.{cand.db_column}"
            if key not in unique:
                unique[key] = cand
            else:
                # 保留分数更高的
                if cand.score > unique[key].score:
                    unique[key] = cand
        
        # 按分数排序
        sorted_candidates = sorted(
            unique.values(),
            key=lambda x: x.score,
            reverse=True
        )
        
        # 只返回前10个
        return sorted_candidates[:10]
```
